function defaultData(){

    return {

        name:"",

        dark:false,

        reminders:{
            morning:false,
            evening:false
        },

        days:{},

        goals:[],

        levelXP:0

    };

}


let data = loadData();


function loadData(){

    try{

        const saved = localStorage.getItem(STORAGE_KEY);

        if(saved){

            return Object.assign(
                defaultData(),
                JSON.parse(saved)
            );

        }

    }catch(e){

        console.error(e);

    }

    return defaultData();

}


function saveData(){

    localStorage.setItem(
        STORAGE_KEY,
        JSON.stringify(data)
    );

}


/* =========================================================
   DATE FUNCTIONS
   ========================================================= */

function dateKey(date=new Date()){

    const y=date.getFullYear();

    const m=String(date.getMonth()+1).padStart(2,"0");

    const d=String(date.getDate()).padStart(2,"0");

    return `${y}-${m}-${d}`;

}


function getDay(key=dateKey()){

    if(!data.days[key]){

        data.days[key]={

            activities:{},

            smallWins:[],

            rating:null,

            reflection:""

        };

    }

    return data.days[key];

}


function formatDate(date=new Date()){

    return date.toLocaleDateString(
        undefined,
        {
            weekday:"long",
            day:"numeric",
            month:"long",
            year:"numeric"
        }
    );

}


/* =========================================================
   GREETING
   ========================================================= */

function updateGreeting(){

    const hour=new Date().getHours();

    let greeting="Good morning";

    if(hour>=12 && hour<18)
        greeting="Good afternoon";

    if(hour>=18)
        greeting="Good evening";

    const name=data.name
        ? " "+data.name
        : "";

    document.getElementById("greeting").innerHTML=
        greeting+" "+name+" 👋";

    document.getElementById("todayDate").textContent=
        formatDate();

}


/* =========================================================
   ACTIVITY GRID
   ========================================================= */

function renderActivities(){

    const container=
        document.getElementById("activityGrid");

    container.innerHTML="";

    const today=getDay();

    Object.keys(activities).forEach(key=>{

        const a=activities[key];

        const done=!!today.activities[key];

        const div=document.createElement("div");

        div.className=
            "activity "+(done?"done":"");

        div.onclick=()=>toggleActivity(key);

        div.innerHTML=`

            <div class="activity-icon">
                ${a.icon}
            </div>

            <div class="activity-name">
                ${a.name}
            </div>

            <div class="activity-desc">
                ${a.desc}
            </div>

            <div class="xp">
                +${a.xp} XP
            </div>

            ${done
                ? `<div class="done-label">✓ Completed</div>`
                : ""
            }

        `;

        container.appendChild(div);

    });

}


/* =========================================================
   ACTIVITY TOGGLE
   ========================================================= */

function toggleActivity(key){

    const today=getDay();

    if(today.activities[key]){

        delete today.activities[key];

        showToast(
            activities[key].name+" unchecked"
        );

    }else{

        today.activities[key]=true;

        showToast(
            activities[key].name+
            " completed · +"+
            activities[key].xp+
            " XP"
        );

    }

    saveData();

    renderAll();

}


/* =========================================================
   XP
   ========================================================= */

function getDayXP(key=dateKey()){

    const day=getDay(key);

    let xp=0;

    Object.keys(day.activities).forEach(a=>{

        if(day.activities[a]){
            xp+=activities[a].xp;
        }

    });

    xp += day.smallWins.length*10;

    return xp;

}

/* =========================================================
   STREAKS
   ========================================================= */

function previousDate(key){

    const d=new Date(key+"T12:00:00");

    d.setDate(d.getDate()-1);

    return dateKey(d);

}


function hasAction(key){

    if(!data.days[key]) return false;

    const d=data.days[key];

    return Object.keys(d.activities||{}).some(
        a=>d.activities[a]
    ) || (d.smallWins||[]).length>0;

}


function hasCheckin(key){

    if(!data.days[key]) return false;

    const d=data.days[key];

    return (
        d.rating!==null ||
        (d.reflection||"").trim()!="" ||
        Object.keys(d.activities||{}).length>0 ||
        (d.smallWins||[]).length>0
    );

}


function calculateStreak(checkin=true){

    let key=dateKey();

    let count=0;

    while(true){

        const exists=
            checkin
            ? hasCheckin(key)
            : hasAction(key);

        if(!exists) break;

        count++;

        key=previousDate(key);

    }

    return count;

}


function updateStreaks(){

    document.getElementById("actionStreak").textContent=
        calculateStreak(false);

    document.getElementById("checkinStreak").textContent=
        calculateStreak(true);

    document.getElementById("level").textContent=
        getLevel();

}

/* =========================================================
   RATING CHART
   ========================================================= */

function renderRatingChart(){

    const canvas=
        document.getElementById("ratingChart");

    const ctx=canvas.getContext("2d");

    const rect=
        canvas.getBoundingClientRect();

    const ratio=
        window.devicePixelRatio || 1;

    canvas.width=
        rect.width*ratio;

    canvas.height=
        rect.height*ratio;

    ctx.scale(ratio,ratio);

    const width=rect.width;

    const height=rect.height;

    ctx.clearRect(0,0,width,height);

    const values=[];

    for(let i=29;i>=0;i--){

        const d=new Date();

        d.setDate(
            d.getDate()-i
        );

        const key=dateKey(d);

        const rating=
            data.days[key]?.rating || null;

        values.push(rating);

    }

    /* grid */

    ctx.strokeStyle=
        getComputedStyle(document.body)
        .getPropertyValue("--border");

    ctx.lineWidth=1;

    for(let y=1;y<=5;y++){

        const yy=
            height -
            ((y-1)/4)*(height-30) -
            15;

        ctx.beginPath();

        ctx.moveTo(30,yy);

        ctx.lineTo(width-10,yy);

        ctx.stroke();

    }

    /* labels */

    ctx.fillStyle=
        getComputedStyle(document.body)
        .getPropertyValue("--muted");

    ctx.font="11px Arial";

    for(let y=1;y<=5;y++){

        const yy=
            height -
            ((y-1)/4)*(height-30) -
            11;

        ctx.fillText(
            y,
            8,
            yy
        );

    }

    /* line */

    let previous=null;

    values.forEach((value,index)=>{

        if(!value){

            previous=null;

            return;

        }

        const x=
            35+
            index*
            ((width-50)/29);

        const y=
            height-
            ((value-1)/4)*(height-30)-
            15;

        if(previous){

            ctx.strokeStyle=
                getComputedStyle(document.body)
                .getPropertyValue("--primary");

            ctx.lineWidth=3;

            ctx.beginPath();

            ctx.moveTo(previous.x,previous.y);

            ctx.lineTo(x,y);

            ctx.stroke();

        }

        ctx.fillStyle=
            getComputedStyle(document.body)
            .getPropertyValue("--primary");

        ctx.beginPath();

        ctx.arc(x,y,4,0,Math.PI*2);

        ctx.fill();

        previous={x,y};

    });

}

function addGoal(){

        document.getElementById("goalTitle")

        );

        document.getElementById("goalPeriod")

        showToast("Enter a valid target.");

        return;

    }

    data.goals.push({

        id:Date.now(),

        title:title,

        target:target,

        progress:0,

        period:period,

        created:new Date().toISOString()

    });

function increaseGoal(id){

    const goal=
        data.goals.find(g=>g.id===id);

    if(!goal) return;

    if(goal.progress<goal.target){

        goal.progress++;

        saveData();

        renderGoals();

        if(goal.progress===goal.target){

            showToast(
                "Goal completed 🎯"
            );

        }else{

            showToast(
                "Goal progress updated."
            );

        }

    }

}


function deleteGoal(id){

    if(!confirm("Delete this goal?")) return;

    data.goals=
        data.goals.filter(
            g=>g.id!==id
        );

    saveData();

    renderGoals();

function exportData(){

    const json=
        JSON.stringify(
            data,
            null,
            2
        );

    const blob=
        new Blob(
            [json],
            {type:"application/json"}
        );

    const url=
        URL.createObjectURL(blob);

    const a=
        document.createElement("a");

    const today=
        dateKey();

    a.href=url;

    a.download=
        `UPWARD_BACKUP_${today}.json`;

    a.click();

    URL.revokeObjectURL(url);

    showToast("Backup exported.");

}


function importData(event){

    const file=
        event.target.files[0];

    if(!file) return;

    const reader=
        new FileReader();

    reader.onload=function(e){

        try{

            const imported=
                JSON.parse(e.target.result);

            if(!imported.days ||
               !imported.goals){

                throw new Error(
                    "Invalid UPWARD backup"
                );

            }

            if(!confirm(
                "Import this backup? Existing data will be replaced."
            )) return;

            data=imported;

            saveData();

            location.reload();

        }catch(err){

            showToast(
                "This does not appear to be a valid UPWARD backup."
            );

        }

    };

    reader.readAsText(file);

}

/* =========================================================
   SECURITY / HTML ESCAPE
   ========================================================= */

function escapeHTML(text){

    return String(text)
        .replace(/&/g,"&amp;")
        .replace(/</g,"&lt;")
        .replace(/>/g,"&gt;")
        .replace(/"/g,"&quot;")
        .replace(/'/g,"&#039;");

}
